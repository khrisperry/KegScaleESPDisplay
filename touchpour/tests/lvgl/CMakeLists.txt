cmake_minimum_required(VERSION 3.16)
project(touchpour_button_state_test C CXX)
if(NOT LVGL_SOURCE_DIR)
    message(FATAL_ERROR "Pass -DLVGL_SOURCE_DIR=/path/to/lvgl-9.5.0")
endif()
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_LIST_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${LVGL_SOURCE_DIR}" lvgl)
add_executable(button_state_test button_state_test.cpp)
target_include_directories(button_state_test PRIVATE "../../main")
target_link_libraries(button_state_test PRIVATE lvgl m)
enable_testing()
add_test(NAME rendered_button_state_changes COMMAND button_state_test)

add_executable(pin_pad_test pin_pad_test.cpp)
target_include_directories(pin_pad_test PRIVATE "../../main")
target_link_libraries(pin_pad_test PRIVATE lvgl m)
add_test(NAME numeric_pin_pad COMMAND pin_pad_test)

# Compile the real screen-flow code against LVGL and a simulated PIN service.
file(READ "${CMAKE_CURRENT_LIST_DIR}/../../main/ui.cpp" pin_ui)
string(FIND "${pin_ui}" "enum class PinMode" pin_start)
string(FIND "${pin_ui}" "void text_if_changed(" pin_end)
math(EXPR pin_length "${pin_end}-${pin_start}")
string(SUBSTRING "${pin_ui}" ${pin_start} ${pin_length} pin_flow)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/pin_flow.inc" "${pin_flow}")
add_executable(pin_flow_test pin_flow_test.cpp)
target_include_directories(pin_flow_test PRIVATE "../../main" "../../components/tap_control" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(pin_flow_test PRIVATE lvgl m)
add_test(NAME pin_configuration_unlock_flow COMMAND pin_flow_test)

add_executable(split_pour_test split_pour_test.cpp)
target_include_directories(split_pour_test PRIVATE "../../main")
target_link_libraries(split_pour_test PRIVATE lvgl m)
add_test(NAME split_pour_hit_targets COMMAND split_pour_test)
